# 合并持仓配置代码-申万&wind_新

## 查询目的
查询X产品在指定时间区间内的收益相关信息。由于表的权限不同，可能得到的产品结果范围会有所不同。

## SQL查询
```sql
/* 名称：查询X产品x时间区间的收益相关信息*/
/* 说明：由于表的权限不同，可能得到的产品结果范围不同*/
/* 编写人员：刘雨*/
/* 联系方式：*/
/* 编写日期：20260105*/

with hld as( 
SELECT a.prod_srrg_key
      ,a.prod_cd
      ,a.vltn_dt
      ,a.scrt_typ_cd
      ,a.scrt_typ_nm
      ,a.scrt_dtl_typ_cd
      ,a.scrt_dtl_typ_nm
      ,a.scrt_srrg_key
      ,a.scrt_wind_cd
      ,a.scrt_chn_abbr
      ,SUM(a.rate) / 100 AS fnct_crrn_mrkt_vl_net_vl_rt
      ,ROW_NUMBER() OVER(PARTITION BY a.vltn_dt, a.prod_srrg_key, a.prod_cd ORDER BY SUM(a.rate) DESC) AS RANK
  FROM (SELECT a.prod_srrg_key
              ,a.prod_cd
              ,a.vltn_dt
              ,a.scrt_typ_cd
              ,replace(a20.enmr_nm,'品种','') AS scrt_typ_nm
              ,a.scrt_dtl_typ_cd
              ,substring(a30.enmr_nm,instr(to_char(a30.enmr_nm),'_') + 1) AS scrt_dtl_typ_nm
              ,a.scrt_srrg_key
              ,a.scrt_wind_cd
              ,a.scrt_chn_abbr
              ,a.hld_qntt
              ,a.fnct_crrn_mkt_vl_rt * 100 AS rate
          FROM dtlab_r.v_dwd_inv_prod_stnd_vltn a
          LEFT JOIN dtlab_r.v_para_stnd_cd a20
            ON a.scrt_typ_cd = a20.enmr_cd AND
               a20.enmr_fld = 'scrt_typ_cd'
          LEFT JOIN dtlab_r.v_para_stnd_cd a30
            ON a.scrt_dtl_typ_cd = a30.enmr_cd AND
               a30.enmr_fld = 'scrt_dtl_typ_cd'
         WHERE a.vltn_dt = '20251031' AND
               a.prod_cd NOT IN ('001371', '100061') AND
               a.scrt_typ_cd IN ('GP', 'ZQ') AND
               a.hld_qntt <> 0) a
 GROUP BY a.prod_srrg_key
         ,a.prod_cd
         ,a.vltn_dt
         ,a.scrt_typ_cd
         ,a.scrt_typ_nm
         ,a.scrt_dtl_typ_cd
         ,a.scrt_dtl_typ_nm
         ,a.scrt_srrg_key
         ,a.scrt_wind_cd
         ,a.scrt_chn_abbr
 ORDER BY a.vltn_dt
         ,a.prod_cd
         ,RANK
),

hld_hk as (
SELECT a.prod_srrg_key
      ,a.prod_cd
      ,a.vltn_dt
      ,a.scrt_typ_cd
      ,a.scrt_typ_nm
      ,a.scrt_dtl_typ_cd
      ,a.scrt_dtl_typ_nm
      ,a.scrt_srrg_key
      ,a.scrt_wind_cd
      ,a.scrt_chn_abbr
      ,SUM(a.rate) / 100 AS fnct_crrn_mrkt_vl_net_vl_rt
      ,ROW_NUMBER() OVER(PARTITION BY a.vltn_dt, a.prod_srrg_key, a.prod_cd ORDER BY SUM(a.rate) DESC) AS RANK
  FROM (SELECT a.prod_srrg_key
              ,a.prod_cd
              ,a.vltn_dt
              ,a.scrt_typ_cd
              ,replace(a20.enmr_nm,'品种','') AS scrt_typ_nm
              ,a.scrt_dtl_typ_cd
              ,substring(a30.enmr_nm,instr(to_char(a30.enmr_nm),'_') + 1) AS scrt_dtl_typ_nm
              ,a.scrt_srrg_key
              ,a.scrt_wind_cd
              ,a.scrt_chn_abbr
              ,a.hld_qntt
              ,a.fnct_crrn_mkt_vl_rt * 100 AS rate
          FROM dtlab_r.v_dwd_inv_prod_stnd_vltn a
          LEFT JOIN dtlab_r.v_para_stnd_cd a20
            ON a.scrt_typ_cd = a20.enmr_cd AND
               a20.enmr_fld = 'scrt_typ_cd'
          LEFT JOIN dtlab_r.v_para_stnd_cd a30
            ON a.scrt_dtl_typ_cd = a30.enmr_cd AND
               a30.enmr_fld = 'scrt_dtl_typ_cd'
         WHERE a.vltn_dt = '20251031' AND
               a.prod_cd IN ('001371', '100061') AND
               a.scrt_typ_cd IN ('GP', 'ZQ') AND
               a.hld_qntt <> 0) a
 GROUP BY a.prod_srrg_key
         ,a.prod_cd
         ,a.vltn_dt
         ,a.scrt_typ_cd
         ,a.scrt_typ_nm
         ,a.scrt_dtl_typ_cd
         ,a.scrt_dtl_typ_nm
         ,a.scrt_srrg_key
         ,a.scrt_wind_cd
         ,a.scrt_chn_abbr
 ORDER BY a.vltn_dt
         ,a.prod_cd
         ,RANK
),

sw_ind as (
SELECT a.s_info_windcode 股票代码
      ,b.s_info_name 股票简称
      ,SUBSTR(a.sw_ind_code, 1, 4) || '000000' 一级行业分类代码
      ,c.industriesname as 一级行业分类名称
  FROM dtlab_r.v_s081_pdata_ashareswnindustriesclass a
  LEFT JOIN dtlab_r.v_s081_pdata_asharedescription b
    ON a.s_info_windcode = b.s_info_windcode
  LEFT JOIN dtlab_r.v_s081_pdata_ashareindustriescode c
    ON SUBSTR(a.sw_ind_code, 1, 4) || '000000' = SUBSTR(c.industriescode, 1, 10)
 WHERE a.cur_sign = '1'
),

wind_ind as (
SELECT a.s_info_windcode 股票代码
      ,b.s_info_name 股票简称
      ,SUBSTR(a.wind_ind_code, 1, 4) || '000000' 一级行业分类代码
      ,industriesname as 一级行业分类名称
  FROM dtlab_r.v_s081_pdata_ashareindustriesclass a
  LEFT JOIN dtlab_r.v_s081_pdata_asharedescription b
    ON a.s_info_windcode = b.s_info_windcode
  LEFT JOIN dtlab_r.v_s081_pdata_ashareindustriescode c
    ON SUBSTR(a.wind_ind_code, 1, 4) || '000000' = SUBSTR(c.industriescode, 1, 10)
 WHERE a.cur_sign = '1'
),

wind_ind_hk as (
SELECT a.s_info_windcode 股票代码
      ,b.s_info_name 股票简称
      ,SUBSTR(a.wind_ind_code, 1, 4) || '000000' 一级行业分类代码
      ,c.industriesname as 一级行业分类名称
  FROM dtlab_r.v_s081_pdata_hkstockwindindustriesmembers a
  LEFT JOIN dtlab_r.v_s081_pdata_hksharedescription b
    ON a.s_info_windcode = b.s_info_windcode
  LEFT JOIN dtlab_r.v_s081_pdata_ashareindustriescode c
    ON SUBSTR(a.wind_ind_code, 1, 4) || '000000' = SUBSTR(c.industriescode, 1, 10)
 WHERE a.cur_sign = '1'
)

SELECT a.vltn_dt
      ,a.prod_cd
      ,a.scrt_typ_nm
      ,a.type
      ,SUM(a.fnct_crrn_mrkt_vl_net_vl_rt) AS fnct_crrn_amnt
      ,ROW_NUMBER() OVER(PARTITION BY a.vltn_dt, a.prod_cd, a.scrt_typ_nm ORDER BY SUM(a.fnct_crrn_mrkt_vl_net_vl_rt) DESC) AS RANK
  FROM (SELECT a.vltn_dt
              ,a.prod_cd
              ,a.scrt_typ_nm
              ,a.scrt_dtl_typ_nm
              ,a.scrt_wind_cd
              ,a.scrt_chn_abbr
              ,a.fnct_crrn_mrkt_vl_net_vl_rt
              ,b.一级行业分类名称
              ,(CASE
                   WHEN a.scrt_typ_cd = 'GP' THEN
                    b.一级行业分类名称
                   WHEN a.scrt_typ_cd = 'ZQ' THEN
                    a.scrt_dtl_typ_nm
                   ELSE
                    a.scrt_typ_nm
               END) AS TYPE
          FROM hld a
          LEFT JOIN sw_ind b
            ON a.scrt_wind_cd = b.股票代码) a
 WHERE a.type IS NOT NULL
 GROUP BY a.vltn_dt
         ,a.prod_cd
         ,a.scrt_typ_nm
         ,a.type
union all 
SELECT a.vltn_dt
      ,a.prod_cd
      ,a.scrt_typ_nm
      ,a.type
      ,SUM(a.fnct_crrn_mrkt_vl_net_vl_rt) AS fnct_crrn_amnt
      ,ROW_NUMBER() OVER(PARTITION BY a.vltn_dt, a.prod_cd, a.scrt_typ_nm ORDER BY SUM(a.fnct_crrn_mrkt_vl_net_vl_rt) DESC) AS RANK
  FROM (SELECT a.vltn_dt
              ,a.prod_cd
              ,a.scrt_typ_nm
              ,a.scrt_dtl_typ_nm
              ,a.scrt_wind_cd
              ,a.scrt_chn_abbr
              ,a.fnct_crrn_mrkt_vl_net_vl_rt
              ,b.一级行业分类名称
              ,c.一级行业分类名称
              ,case
                when a.scrt_typ_cd = 'GP' and right(a.scrt_wind_cd,2) != 'HK' then b.一级行业分类名称
                when a.scrt_typ_cd = 'GP' and right(a.scrt_wind_cd,2) = 'HK' then c.一级行业分类名称
                when a.scrt_typ_cd = 'GP' then a.scrt_dtl_typ_nm
              else a.scrt_typ_nm end as type
          FROM hld_hk a
          LEFT JOIN wind_ind b
            ON a.scrt_wind_cd = b.股票代码
          LEFT JOIN wind_ind_hk c
            ON a.scrt_wind_cd = c.股票代码) a
 WHERE a.type IS NOT NULL
 GROUP BY a.vltn_dt
         ,a.prod_cd
         ,a.scrt_typ_nm
         ,a.type
 ORDER BY vltn_dt
         ,scrt_typ_nm
         ,RANK
;
```

## 查询说明
此查询用于分析特定产品在指定时间区间内的市场价值净利率，并按类型和排名进行汇总。查询中使用了多个CTE（Common Table Expressions）来分别处理不同的数据集，并最终将这些数据集合并以提供全面的分析结果。

## 条件说明
- `vltn_dt`：估值日期，查询条件为 '20251031'。
- `prod_cd`：产品代码，排除了 '001371' 和 '100061'。
- `scrt_typ_cd`：证券类型代码，仅包括 'GP'（股票）和 'ZQ'（债券）。
- `hld_qntt`：持仓数量，排除了持仓数量为0的情况。

## 业务场景
此查询适用于投资组合分析，特别是需要了解特定产品在特定时间区间内的收益情况及其在不同行业或证券类型中的分布。
