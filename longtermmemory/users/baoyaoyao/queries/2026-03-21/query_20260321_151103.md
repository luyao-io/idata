# 合并】业绩归因代码-申万&wind_新

## 查询目的
查询X产品在指定时间区间内的收益相关信息，包括产品的证券类型、证券详细类型、证券Wind代码、证券简称、功能当前金额等，并进行相应的行业分类。

## SQL查询
```sql
/* 名称：查询X产品x时间区间的收益相关信息*/
/* 说明：由于表的权限不同，可能得到的产品结果范围不同*/
/* 编写人员：刘雨*/
/* 联系方式：*/
/* 编写日期：20260105*/

with vchr as( 
SELECT a1.prod_cd
      ,a1.scrt_typ_cd
      ,REPLACE(a10.enmr_nm, '品种', '') AS scrt_typ_nm
      ,a1.scrt_dtl_typ_cd
      ,a20.enmr_nm AS scrt_dtl_typ_nm
      ,a1.scrt_wind_cd
      ,a1.scrt_chn_abbr
      ,SUM(a1.fnct_crrn_trnv_amnt) AS fnct_crrn_amnt
  FROM dtlab_r.v_dwd_inv_prod_stnd_vchr a1
  LEFT JOIN dtlab_r.v_dim_prod_stnd_itm_dmns a2
    ON a1.stnd_itm_srrg_key = a2.stnd_itm_srrg_key
  LEFT JOIN dtlab_r.v_para_stnd_cd a10
    ON a1.scrt_typ_cd = a10.enmr_cd AND
       a10.enmr_fld = 'scrt_typ_cd'
  LEFT JOIN dtlab_r.v_para_stnd_cd a20
    ON a1.scrt_dtl_typ_cd = a20.enmr_cd AND
       a20.enmr_fld = 'scrt_dtl_typ_cd'
 WHERE  a2.itm_typ_cd = 'SY' AND /*科目类别：损益类*/
       COALESCE(a1.acct_bsns_itm_cd, '') NOT LIKE '%SYJZ%' AND /*排除结转收益*/
       a1.ln_drct_cd = 'D' AND /*借贷方向*/
       a1.vchr_dt BETWEEN '20251001' AND '20251031' AND
       a1.prod_cd NOT IN ('001371', '100061', '001345') AND
       a2.acct_proj_cd IN (
        /*差价收入 口径*/
       'TZSY_CJSR' /*投资收益_差价收入*/ ,'TZSY_CJSR_XXS' /*投资收益_差价收入_销项税*/ ,'QTZHSY_CJSR' /*其他综合收益_差价收入*/ ,'TZSY_ZZS_CD' /*投资收益_增值税_冲抵*/ ,'TZSY_TDSY' /*投资收益_替代损益*/
       /*利息收入 口径*/
       ,'LXSR_ZJ' /*利息收入_资金*/ ,'LXSR_ZRQ_XXS' /*利息收入_转融券_销项税*/ ,'LXSR_ZJ_XXS' /*利息收入_资金_销项税*/ ,'LXSR_ZQ' /*利息收入_证券*/ ,'LXSR_ZQ_XXS' /*利息收入_证券_销项税*/ ,'LXSR_ZRZ_XXS' /*利息收入_转融资_销项税*/ ,'LXSR' /*利息收入*/
       ,'LXSR_TA' /*利息收入_销售*/  ,'LXSR_ZRQ' /*利息收入_转融券*/ ,'LXSR_ZRZ' /*利息收入_转融资*/ ,'LXZC_RZ' /*利息支出_融资*/ ,'LXZC_ZJ' /*利息支出_资金*/ ,'LXZC_ZQ' /*利息支出_证券*/ ,'LXZC' /*利息支出*/ ,'LXZC_RQ' /*利息支出_融券*/
       /*红利收入 口径*/
       ,'TZSY_HLSR' /*投资收益_红利收入*/, 'TZSY_HLSR_XXS' /*投资收益_红利收入_销项税*/
       )
 GROUP BY a1.prod_cd
         ,a1.scrt_typ_cd
         ,a10.enmr_nm
         ,a1.scrt_dtl_typ_cd
         ,a20.enmr_nm
         ,a1.scrt_wind_cd
         ,a1.scrt_chn_abbr
union all 
/*估增口径*/        
SELECT a1.prod_cd
      ,a1.scrt_typ_cd
      ,REPLACE(a10.enmr_nm, '品种', '') AS scrt_typ_nm
      ,a1.scrt_dtl_typ_cd
      ,a20.enmr_nm AS scrt_dtl_typ_nm
      ,a1.scrt_wind_cd
      ,a1.scrt_chn_abbr
      ,SUM(a1.fnct_crrn_trnv_amnt) AS fnct_crrn_amnt
  FROM dtlab_r.v_dwd_inv_prod_stnd_vchr a1
  LEFT JOIN dtlab_r.v_dim_prod_stnd_itm_dmns a2
    ON a1.stnd_itm_srrg_key = a2.stnd_itm_srrg_key
  LEFT JOIN dtlab_r.v_para_stnd_cd a10
    ON a1.scrt_typ_cd = a10.enmr_cd AND
       a10.enmr_fld = 'scrt_typ_cd'
  LEFT JOIN dtlab_r.v_para_stnd_cd a20
    ON a1.scrt_dtl_typ_cd = a20.enmr_cd AND
       a20.enmr_fld = 'scrt_dtl_typ_cd'
 WHERE  a2.itm_typ_cd = 'SY' AND /*科目类别：损益类*/
       COALESCE(a1.acct_bsns_itm_cd, '') NOT LIKE '%SYJZ%' AND /*排除结转收益*/
       a1.ln_drct_cd = 'D' AND /*借贷方向*/
       a1.vchr_dt BETWEEN '20251001' AND '20251031' AND
       a1.prod_cd NOT IN ('001371', '100061', '001345') AND
       ((a2.itm_typ_cd = 'SY' /*科目类别：损益类*/ AND a2.acct_proj_cd in ('GYBD','GYBD_KTTD'))/*核算项目：公允价值变动损益*/ OR 
       (a2.itm_typ_cd = 'QY' /*科目类别：权益类*/ AND a2.acct_proj_cd = 'ZBGJ' /*核算项目：资本公积*/AND a1.invs_clss_cd IN ('FS')/*可供出售类*/)  OR 
       (a1.prod_cd in ('YL0502','SB0114') and (a2.itm_typ_cd = 'QY' /*科目类别：权益类*/ AND a2.acct_proj_cd = 'LRFP_WFP_WSX_ZZZH' /*核算项目：利润分配_未分配利润_未实现(准则转换)*/)   /*20250409 liuyu*/ ) )
 GROUP BY a1.prod_cd
         ,a1.scrt_typ_cd
         ,a10.enmr_nm
         ,a1.scrt_dtl_typ_cd
         ,a20.enmr_nm
         ,a1.scrt_wind_cd
         ,a1.scrt_chn_abbr
),

sw_ind as (
SELECT a.s_info_windcode 股票代码
      ,b.s_info_name 股票简称
      ,SUBSTR(a.sw_ind_code, 1, 4) || '000000' 一级行业分类代码
      ,c.industriesname as 一级行业分类名称
  FROM dtlab_r.v_s081_pdata_ashareswnindustriesclass a
  LEFT JOIN dtlab_r.v_s081_pdata_asharedescription b
    ON a.s_info_windcode = b.s_info_windcode
  LEFT JOIN dtlab_r.v_s081_pdata_ashareindustriescode c
    ON SUBSTR(a.sw_ind_code, 1, 4) || '000000' = SUBSTR(c.industriescode, 1, 10)
 WHERE a.cur_sign = '1'
),

wind_ind as (
SELECT a.s_info_windcode 股票代码
      ,b.s_info_name 股票简称
      ,SUBSTR(a.wind_ind_code, 1, 4) || '000000' 一级行业分类代码
      ,industriesname as 一级行业分类名称
  FROM dtlab_r.v_s081_pdata_ashareindustriesclass a
  LEFT JOIN dtlab_r.v_s081_pdata_asharedescription b
    ON a.s_info_windcode = b.s_info_windcode
  LEFT JOIN dtlab_r.v_s081_pdata_ashareindustriescode c
    ON SUBSTR(a.wind_ind_code, 1, 4) || '000000' = SUBSTR(c.industriescode, 1, 10)
 WHERE a.cur_sign = '1'
),

wind_ind_hk as (
SELECT a.s_info_windcode 股票代码
      ,b.s_info_name 股票简称
      ,SUBSTR(a.wind_ind_code, 1, 4) || '000000' 一级行业分类代码
      ,c.industriesname as 一级行业分类名称
  FROM dtlab_r.v_s081_pdata_hkstockwindindustriesmembers a
  LEFT JOIN dtlab_r.v_s081_pdata_hksharedescription b
    ON a.s_info_windcode = b.s_info_windcode
  LEFT JOIN dtlab_r.v_s081_pdata_ashareindustriescode c
    ON SUBSTR(a.wind_ind_code, 1, 4) || '000000' = SUBSTR(c.industriescode, 1, 10)
 WHERE a.cur_sign = '1'
)
 

SELECT a.prod_cd
      ,a.scrt_typ_nm
      ,a.type
      ,SUM(a.fnct_crrn_amnt) AS fnct_crrn_amnt
      ,(CASE WHEN SUM(a.fnct_crrn_amnt) >= 0 THEN ROW_NUMBER() OVER(PARTITION BY a.prod_cd, a.scrt_typ_nm ORDER BY SUM(a.fnct_crrn_amnt) DESC) ELSE NULL END) AS RANK
      ,(CASE WHEN SUM(a.fnct_crrn_amnt) < 0 THEN ROW_NUMBER() OVER(PARTITION BY a.prod_cd, a.scrt_typ_nm ORDER BY - SUM(a.fnct_crrn_amnt) DESC) ELSE NULL END) AS n_rank
  FROM (SELECT a.prod_cd
              ,a.scrt_typ_nm
              ,a.scrt_dtl_typ_nm
              ,a.scrt_wind_cd
              ,a.scrt_chn_abbr
              ,a.fnct_crrn_amnt
              ,b.一级行业分类名称
              ,(CASE
                   WHEN a.scrt_typ_cd = 'GP' THEN b.一级行业分类名称
                   WHEN a.scrt_typ_cd = 'ZQ' THEN a.scrt_dtl_typ_nm
                   ELSE a.scrt_typ_nm END) AS TYPE
          FROM vchr a
          LEFT JOIN sw_ind b
            ON a.scrt_wind_cd = b.股票代码) a
 WHERE a.type IS NOT NULL
 GROUP BY a.prod_cd
         ,a.scrt_typ_nm
         ,a.type
```

## 查询说明
此查询用于分析特定时间段内（2025年10月1日至2025年10月31日）X产品的收益情况。通过连接多个表，获取产品的证券类型、证券详细类型、证券Wind代码、证券简称、功能当前金额等信息，并根据不同的证券类型和行业分类进行汇总和排名。

## 条件说明
- `vchr` 子查询中，筛选了科目类别为“损益类”且借贷方向为“借方”的记录，并排除了结转收益的记录。
- `sw_ind` 和 `wind_ind` 子查询分别获取了A股和港股的行业分类信息。
- 最终的查询结果中，对每个产品的证券类型和行业分类进行了汇总，并根据收益金额进行了正向和负向排名。

## 业务场景
此查询适用于投资组合分析，帮助投资者或分析师了解特定产品在特定时间段内的收益表现，以及其在不同行业中的表现情况。
