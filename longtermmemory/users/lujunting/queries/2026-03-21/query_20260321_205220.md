select
  a.clnd_dt as '自然日期',
  a.fnd_acct as '基金账号',
  a.trd_acct as '交易账号',
  a.cstm_fll_nm as '账户名称',
  a.cstm_typ as '客户类型',
  a.mn_prod_nm as '主产品名称',
  a.mn_prod_cd as '主产品代码',
  a.prod_nm as '产品名称',
  a.prod_cd as '产品代码',

  a.prod_typ_200272 as '公募产品类型',
  case 
  when t1.syqz is null then a.shr_blnc
  else a.shr_blnc*t1.syqz/100
  end as '确认份额' ,
  case 
  when t1.syqz is null then a.amnt_blnc_4
  else a.amnt_blnc_4*t1.syqz/100 end as '确认金额',
  t1.syqz as '收益权重',
  d.qdjl as '渠道经理',

  t3.name as '团队',
  t4.name as '部门',

  a.sll_chnn_cd_0 as '销售渠道代码',
  a.sll_chnn_abbr_0 as '渠道名称',
  a.sll_chnn_typ as '渠道类型',
  c.GSMC as '客户名称',
  case
    when b.ZHMC is null then '渠道'
    else '机构'
  end as 'CRM业务类型'
from
  dtlab_r.v_dws_mkt_all_chnn_hld_indx a
  left join dtlab_r.v_s060_crm_r_vecif_jgzhlb b on b.jyzh = a.trd_acct

  left join dtlab_r.v_s060_ecif_tjj_jgkhgx t1 on b.id =t1.khh and t1.zt='0'
  left join dtlab_r.v_s060_ecif_tjj_jgkhkzxx_jgkh c on c.id = b.khgs
  left join dtlab_w.sales_channel_data d on d.qddm=a.sll_chnn_cd_0
  left join dtlab_r.v_s060_jjcrm_tuser t2 on decode(t1.ryxx,null,d.qdjl,t1.ryxx) = t2.id
  left join dtlab_r.v_s060_jjcrm_lborganization t3 on t2.orgid = t3.id
  left join dtlab_r.v_s060_jjcrm_lborganization t4 on t3.fid = t4.id
where
  a.clnd_dt >= date ('2023-12-04')
  and a.clnd_dt <= date ('2023-12-04')
  and a.prod_typ_203427='公募'

## 查询说明
此查询用于从多个视图中提取特定日期范围内的公募产品持有信息，包括但不限于基金账号、交易账号、账户名称、客户类型等，并通过多个左连接（LEFT JOIN）关联其他相关表以获取更详细的信息，如渠道经理、销售、团队和部门等。

## 条件说明
- `a.clnd_dt >= date ('2023-12-04') and a.clnd_dt <= date ('2023-12-04')`: 指定查询的日期范围为2023年12月4日。
- `a.prod_typ_203427='公募'`: 仅选择产品类型为“公募”的记录。
