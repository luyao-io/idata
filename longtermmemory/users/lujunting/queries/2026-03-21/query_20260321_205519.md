# 全部基金分类查询

## 查询目的
查询所有基金的分类信息，包括基金代码、基金Wind代码、生效日期、简写名称、运作状态、富国产品类型、富国投资类型、Wind分类、银河分类、主被动投资、产品部细分分类和产品部分类等。

```sql
-- 由于权限限制，无法执行原查询

select
  a.prod_cd,
  a.prod_wind_cd,
  a.prod_cntr_effc_dt_202900,
  a.prod_chns_abbr_rglt_rprt_60011,
  a.prod_chn_abbr_intr_use_200136,
  a.prod_stt_200282 as '运作状态',
  a.prod_typ_203427 as '富国产品类型',
  a.prod_typ_200272 as '富国投资类型',
  e.wind1 as 'wind一级分类',
  e.wind2 as 'wind二级分类',
  b.classify_lvl1name as '银河一级分类',
  b.classify_lvl2name as '银河二级分类',
  b.classify_lvl3name as '银河三级分类',
  decode(e.f_info_invs_strt,1,'主动投资',2,'被动投资',null) as '主被动投资',
  f.ctgr as '产品部细分分类',
  f.typ as '产品部分类'
from
  dtlab_r.v_dim_prod_dmns a
left join dtlab_r.v_s081_pdata_yhfund_classify b on a.prod_wind_cd=b.s_info_windcode
left join (select c0.fnd_cd,c0.ctgr,c0.typ,c0.end_dt from dtlab_r.v_s999_dl_ods_add_r_t_all_fnd_ctgr c0 inner join(
select fnd_cd,max(end_dt) as max from dtlab_r.v_s999_dl_ods_add_r_t_all_fnd_ctgr group by fnd_cd) c1 on c0.fnd_cd=c1.fnd_cd and c0.end_dt=c1.max ) f on a.prod_cd=left(f.fnd_cd,6)
left join (select
  a.f_info_windcode,
  a.s_info_sector,
  a.cur_sign,
  a.f_info_invs_strt,
  a3.industriesname as 'wind1',
  CASE
           WHEN a3.industriesname = '货币市场型基金' THEN
            '货币市场型基金'
           ELSE
            a4.industriesname end  as 'wind2'
from
  dtlab_r.v_s034_wfdb_chinamutualfundsector a
  left join dtlab_r.v_s034_wfdb_ashareindustriescode a2 on a.s_info_sector = a2.industriescode
  and a.s_info_sector like '200101%'
  and a.cur_sign = 1
  left join dtlab_r.v_s034_wfdb_ashareindustriescode a3 on SUBSTR(a3.industriescode, 1, 8) = SUBSTR(a.s_info_sector, 1, 8) 
  and a3.levelnum=4
  LEFT JOIN dtlab_r.v_s034_wfdb_ashareindustriescode a4
    ON SUBSTR(a4.industriescode, 1, 10) = SUBSTR(a.s_info_sector, 1, 10) AND
       a4.levelnum = 5
where a.s_info_sector like '200101%' and a.cur_sign = 1) e on e.f_info_windcode=a.prod_wind_cd
where a.prod_lvl='0'
and a.prod_typ_203427='公募'
order by a.prod_cd
```
