```sql
-- 由于权限限制，无法执行原查询
-- 原查询意图：查询A500ETF产品的销售峰值规模
select K.销售,
  max (k.规模) as '峰值'
  from (
  select m.日期 as '日期',
  m.销售 as '销售',
  sum(m.规模) as '规模'
  from (
   select
  clnd_dt AS '日期',
  crm_bsns_typ,
  crm_cstm_nm,
  case
    when t1.syqz is null then 1
    else t1.syqz / 100
  end as '收益权重',
  sum(tdy_hld_amnt * (case when t1.syqz is null then 1 else t1.syqz / 100 end)) as '规模',
  t2.name as '销售'
from
  dtlab_r.v_t_orgn_cstm_prft_dat a
  left join dtlab_r.v_s060_crm_r_vecif_jgzhlb b on b.jyzh = a.trd_acct
  and b.xsjg = a.chnn_cd
  left join dtlab_r.v_s060_ecif_tjj_jgkhgx t1 on b.id = t1.khh
  and t1.zt = '0'
  left join dtlab_r.v_s060_jjcrm_tuser t2 on t1.ryxx = t2.id
  left join dtlab_r.v_s060_jjcrm_lborganization t3 on t2.orgid = t3.id
  left join dtlab_r.v_s060_jjcrm_lborganization t4 on t3.fid = t4.id
where
  a.clnd_dt between '2024-12-01' and '2024-12-31'
  and b.zhmc is not null
  and prod_cd in ('022463', '022464', '022902', '563220')
group by
  clnd_dt,
  crm_bsns_typ,
  crm_cstm_nm,
  t2.name,
  t1.syqz) m
  group by m.日期,
  m.销售) k
  group by k.销售
```
